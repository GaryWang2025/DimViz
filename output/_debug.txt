function colorFromW(w) {
      const t = (w + 2) / 4;
      const r = Math.round(40 + 215 * t);
      const g = Math.round(80 + 60 * (1 - Math.abs(t - 0.5) * 2));
      const b = Math.round(220 - 180 * t);
      return new THREE.Color(r/255, g/255, b/255);
    }
